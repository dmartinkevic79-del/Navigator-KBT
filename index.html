function doGet(e) { var callback = e.parameter.callback; var inputPassword = e.parameter.password; var result = { success: false, user: "" }; try { var sheetId = '1FPHPwKeBmFrMqp2YWTmgN8xvrlsVaKZrrXd5oe8KoWg'; var sheet = SpreadsheetApp.openById(sheetId).getSheetByName('Доступы'); if (sheet) { var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { var sheetName = data[i][0] ? data[i][0].toString().trim() : ""; var sheetPassword = data[i][1] ? data[i][1].toString().trim() : ""; var status = data[i][2] ? data[i][2].toString().trim() : ""; if (sheetPassword === inputPassword && inputPassword !== "" && status === "Активен") { result.success = true; result.user = sheetName; // Записываем время входа в столбец D var now = new Date(); var formattedDate = Utilities.formatDate(now, Session.getScriptTimeZone(), "dd.MM.yyyy HH:mm:ss"); sheet.getRange(i + 1, 4).setValue(formattedDate); break; } } } } catch (err) { result.error = err.toString(); } var jsonString = JSON.stringify(result); if (callback) { jsonString = callback + "(" + jsonString + ")"; } return ContentService.createTextOutput(jsonString) .setMimeType(ContentService.MimeType.JAVASCRIPT); }
